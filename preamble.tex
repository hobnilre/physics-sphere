\allowdisplaybreaks

% Text that continues after a displayed equation should not be indented.
% Pandoc necessarily starts a new paragraph there, so strip the indent box
% at the start of the paragraph that follows any display.
\usepackage{etoolbox}
\makeatletter
\newcommand*{\ph@noindentnext}{\everypar{{\setbox\z@\lastbox}\everypar{}}}
\AfterEndEnvironment{equation}{\ph@noindentnext}
\AfterEndEnvironment{equation*}{\ph@noindentnext}
\AfterEndEnvironment{align}{\ph@noindentnext}
\AfterEndEnvironment{align*}{\ph@noindentnext}
% Discourage a page break between a lead-in sentence and the display it
% introduces, which Pandoc necessarily separates with \par.
\BeforeBeginEnvironment{equation}{\nopagebreak[4]}
\BeforeBeginEnvironment{equation*}{\nopagebreak[4]}
\makeatother

% Tighter, journal-like lists; symbol definitions set as a nomenclature list.

\setlist[enumerate]{topsep=4pt,itemsep=3pt,parsep=0pt}
\setlist[description]{topsep=5pt,itemsep=3pt,parsep=0pt,
                      labelwidth=1.6em,leftmargin=3.2em,style=sameline,font=\normalfont}

% Abstract set small and inset, as in a journal note.
\renewcommand{\abstractnamefont}{\normalfont\bfseries}
\renewcommand{\abstracttextfont}{\normalfont\small}
\setlength{\absleftindent}{0.05\textwidth}
\setlength{\absrightindent}{0.05\textwidth}

\titlespacing*{\subsection}{0pt}{1.8ex plus 0.5ex minus 0.2ex}{0.9ex plus 0.2ex}
